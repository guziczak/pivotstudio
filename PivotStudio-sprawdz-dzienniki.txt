# Pivot Studio - sprawdzenie dziennikow Windows po zamknieciu 6.10.2026 ok. 15:05
#
# Jak uruchomic (na komputerze, na ktorym Pivot sie zamknal):
#   1. Otworz Windows PowerShell (nie cmd).
#   2. Wklej cala zawartosc tego pliku i nacisnij Enter.
# Polecenie tylko odczytuje dzienniki z 6.10.2026, 14:55-15:15 i niczego nie zmienia.
# Okno czasu jest wpisane na stale, wiec dziala tez pozniejszego dnia.
#
# Jak czytac wynik:
#   - Wpis Defendera (np. 1116, 1117, 1121): widac, co zablokowalo Pivot i dlaczego.
#     Z tym mozna isc prosto do IT.
#   - Blad aplikacji 1000 dla pythonw.exe albo powershell.exe: to jednak awaria programu.
#   - Wszedzie "nie znaleziono zdarzen": najpewniej zablokowalo go zabezpieczenie
#     innej firmy niz Microsoft; jego dzienniki widzi tylko IT.
#   - "Odmowa dostepu" przy ktoryms dzienniku: ten dziennik moze przejrzec tylko IT.
#
# Wynik skopiuj i wklej do rozmowy z Claude.

$od=[datetime]'2026-10-06 14:55'; $do=[datetime]'2026-10-06 15:15'
$logi=[ordered]@{
 'Application'=@{Level=1,2}
 'Microsoft-Windows-Windows Defender/Operational'=@{Id=1006,1007,1008,1015,1116,1117,1118,1119,1121,1122,1123,1124,1125,1126}
 'Microsoft-Windows-PowerShell/Operational'=@{Level=1,2,3}
 'Microsoft-Windows-CodeIntegrity/Operational'=@{Level=1,2,3}
 'Microsoft-Windows-AppLocker/EXE and DLL'=@{Level=1,2,3}
}
foreach($l in $logi.Keys){
 "== $l"
 $f=@{LogName=$l;StartTime=$od;EndTime=$do}+$logi[$l]
 try{Get-WinEvent -FilterHashtable $f -MaxEvents 15 -ErrorAction Stop | ForEach-Object {$m=$_.Message -replace '\s+',' '; if($m.Length -gt 300){$m=$m.Substring(0,300)}; '{0:HH:mm:ss} [{1}] {2}' -f $_.TimeCreated,$_.Id,$m}}
 catch{'   '+$_.Exception.Message}
}
